import tkinter as tk
from tkinter import ttk, messagebox, filedialog
import sqlite3
import csv
import hashlib
from datetime import datetime

# ==========================================
# DATABASE SETUP
# ==========================================

conn = sqlite3.connect("student_database.db")
cursor = conn.cursor()

cursor.execute("""
CREATE TABLE IF NOT EXISTS users (
    username TEXT PRIMARY KEY,
    password TEXT NOT NULL
)
""")

cursor.execute("""
CREATE TABLE IF NOT EXISTS students (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    matric_no TEXT UNIQUE NOT NULL,
    name TEXT NOT NULL,
    gender TEXT,
    department TEXT,
    level TEXT,
    email TEXT,
    phone TEXT
)
""")

# Create a default administrator account
default_password = hashlib.sha256(
    "admin123".encode()
).hexdigest()

cursor.execute("""
INSERT OR IGNORE INTO users (username, password)
VALUES (?, ?)
""", ("admin", default_password))

conn.commit()

# ==========================================
# MAIN APPLICATION
# ==========================================

class StudentManagementSystem:

    def __init__(self, root):
        self.root = root
        self.root.title("Student Database Management System")
        self.root.geometry("1100x650")
        self.root.configure(bg="#eef2f7")

        self.login_screen()

    # ======================================
    # LOGIN SCREEN
    # ======================================

    def login_screen(self):
        self.clear_window()

        frame = tk.Frame(
            self.root,
            bg="white",
            padx=40,
            pady=35
        )
        frame.place(relx=0.5, rely=0.5, anchor="center")

        tk.Label(
            frame,
            text="STUDENT MANAGEMENT SYSTEM",
            font=("Arial", 17, "bold"),
            bg="white",
            fg="#17365d"
        ).pack(pady=15)

        tk.Label(
            frame,
            text="Username",
            bg="white"
        ).pack(anchor="w")

        self.username_entry = tk.Entry(
            frame,
            width=35
        )
        self.username_entry.pack(pady=5)

        tk.Label(
            frame,
            text="Password",
            bg="white"
        ).pack(anchor="w")

        self.password_entry = tk.Entry(
            frame,
            width=35,
            show="*"
        )
        self.password_entry.pack(pady=5)

        tk.Button(
            frame,
            text="LOGIN",
            width=25,
            bg="#17365d",
            fg="white",
            command=self.login
        ).pack(pady=20)

        tk.Label(
            frame,
            text="Default login: admin / admin123",
            bg="white",
            fg="gray"
        ).pack()

        self.password_entry.bind(
            "<Return>",
            lambda event: self.login()
        )

    def login(self):
        username = self.username_entry.get().strip()
        password = self.password_entry.get()

        hashed_password = hashlib.sha256(
            password.encode()
        ).hexdigest()

        cursor.execute("""
        SELECT username FROM users
        WHERE username = ? AND password = ?
        """, (username, hashed_password))

        if cursor.fetchone():
            self.dashboard()
        else:
            messagebox.showerror(
                "Login Failed",
                "Incorrect username or password."
            )

    # ======================================
    # DASHBOARD
    # ======================================

    def dashboard(self):
        self.clear_window()

        header = tk.Frame(
            self.root,
            bg="#17365d",
            height=65
        )
        header.pack(fill="x")

        tk.Label(
            header,
            text="STUDENT DATABASE MANAGEMENT SYSTEM",
            font=("Arial", 17, "bold"),
            bg="#17365d",
            fg="white"
        ).pack(pady=15)

        content = tk.Frame(
            self.root,
            bg="#eef2f7",
            padx=20,
            pady=15
        )
        content.pack(fill="both", expand=True)

        # Student form
        form = tk.LabelFrame(
            content,
            text="Student Information",
            font=("Arial", 11, "bold"),
            bg="white",
            padx=12,
            pady=10
        )
        form.pack(fill="x", pady=5)

        self.fields = {}

        field_names = [
            ("Matric Number", "matric_no"),
            ("Full Name", "name"),
            ("Gender", "gender"),
            ("Department", "department"),
            ("Level", "level"),
            ("Email", "email"),
            ("Phone", "phone")
        ]

        for index, (label, key) in enumerate(field_names):
            row = index // 4
            col = (index % 4) * 2

            tk.Label(
                form,
                text=label,
                bg="white"
            ).grid(
                row=row * 2,
                column=col,
                sticky="w",
                padx=5,
                pady=3
            )

            if key == "gender":
                widget = ttk.Combobox(
                    form,
                    values=["Male", "Female", "Other"],
                    width=19,
                    state="readonly"
                )
            elif key == "level":
                widget = ttk.Combobox(
                    form,
                    values=["100", "200", "300", "400",
                            "500", "600"],
                    width=19,
                    state="readonly"
                )
            else:
                widget = tk.Entry(
                    form,
                    width=22
                )

            widget.grid(
                row=row * 2 + 1,
                column=col,
                padx=5,
                pady=3,
                sticky="w"
            )

            self.fields[key] = widget

        # Buttons
        buttons = tk.Frame(
            content,
            bg="#eef2f7"
        )
        buttons.pack(fill="x", pady=10)

        button_data = [
            ("Add Student", self.add_student),
            ("Search", self.search_student),
            ("Update", self.update_student),
            ("Delete", self.delete_student),
            ("Clear Form", self.clear_form),
            ("Generate Report", self.generate_report),
            ("Logout", self.login_screen)
        ]

        for text, command in button_data:
            tk.Button(
                buttons,
                text=text,
                command=command,
                bg="#17365d",
                fg="white",
                padx=8,
                pady=5
            ).pack(
                side="left",
                padx=3
            )

        # Search box
        search_frame = tk.Frame(
            content,
            bg="#eef2f7"
        )
        search_frame.pack(fill="x", pady=5)

        tk.Label(
            search_frame,
            text="Search by name or matric number:",
            bg="#eef2f7"
        ).pack(side="left")

        self.search_entry = tk.Entry(
            search_frame,
            width=35
        )
        self.search_entry.pack(
            side="left",
            padx=8
        )

        tk.Button(
            search_frame,
            text="View All Students",
            command=self.load_students
        ).pack(side="left")

        # Student table
        table_frame = tk.Frame(content)
        table_frame.pack(
            fill="both",
            expand=True,
            pady=5
        )

        columns = (
            "id",
            "matric_no",
            "name",
            "gender",
            "department",
            "level",
            "email",
            "phone"
        )

        self.tree = ttk.Treeview(
            table_frame,
            columns=columns,
            show="headings"
        )

        headings = {
            "id": "ID",
            "matric_no": "Matric Number",
            "name": "Full Name",
            "gender": "Gender",
            "department": "Department",
            "level": "Level",
            "email": "Email",
            "phone": "Phone"
        }

        for column in columns:
            self.tree.heading(
                column,
                text=headings[column]
            )
            self.tree.column(
                column,
                width=120,
                anchor="center"
            )

        scrollbar = ttk.Scrollbar(
            table_frame,
            orient="vertical",
            command=self.tree.yview
        )

        self.tree.configure(
            yscrollcommand=scrollbar.set
        )

        self.tree.pack(
            side="left",
            fill="both",
            expand=True
        )

        scrollbar.pack(
            side="right",
            fill="y"
        )

        self.tree.bind(
            "<<TreeviewSelect>>",
            self.select_student
        )

        self.load_students()

    # ======================================
    # GET FORM DATA
    # ======================================

    def get_form_data(self):
        return {
            key: widget.get().strip()
            for key, widget in self.fields.items()
        }

    # ======================================
    # ADD STUDENT
    # ======================================

    def add_student(self):
        data = self.get_form_data()

        if not data["matric_no"] or not data["name"]:
            messagebox.showwarning(
                "Missing Information",
                "Matric number and full name are required."
            )
            return

        try:
            cursor.execute("""
            INSERT INTO students (
                matric_no, name, gender, department,
                level, email, phone
            )
            VALUES (?, ?, ?, ?, ?, ?, ?)
            """, (
                data["matric_no"],
                data["name"],
                data["gender"],
                data["department"],
                data["level"],
                data["email"],
                data["phone"]
            ))

            conn.commit()

            messagebox.showinfo(
                "Success",
                "Student added successfully."
            )

            self.clear_form()
            self.load_students()

        except sqlite3.IntegrityError:
            messagebox.showerror(
                "Error",
                "This matric number already exists."
            )

    # ======================================
    # SEARCH STUDENT
    # ======================================

    def search_student(self):
        search_text = self.search_entry.get().strip()

        if not search_text:
            messagebox.showwarning(
                "Search",
                "Enter a name or matric number."
            )
            return

        cursor.execute("""
        SELECT * FROM students
        WHERE name LIKE ? OR matric_no LIKE ?
        ORDER BY id
        """, (
            f"%{search_text}%",
            f"%{search_text}%"
        ))

        self.display_students(cursor.fetchall())

    # ======================================
    # LOAD ALL STUDENTS
    # ======================================

    def load_students(self):
        cursor.execute("""
        SELECT * FROM students ORDER BY id
        """)

        self.display_students(cursor.fetchall())

    def display_students(self, records):
        for item in self.tree.get_children():
            self.tree.delete(item)

        for record in records:
            self.tree.insert(
                "",
                "end",
                values=record
            )

    # ======================================
    # SELECT STUDENT FROM TABLE
    # ======================================

    def select_student(self, event=None):
        selected = self.tree.selection()

        if not selected:
            return

        values = self.tree.item(
            selected[0],
            "values"
        )

        if not values:
            return

        keys = [
            "matric_no",
            "name",
            "gender",
            "department",
            "level",
            "email",
            "phone"
        ]

        for key, value in zip(keys, values[1:]):
            widget = self.fields[key]

            widget.delete(0, tk.END)
            widget.insert(0, value)

    # ======================================
    # UPDATE STUDENT
    # ======================================

    def update_student(self):
        selected = self.tree.selection()

        if not selected:
            messagebox.showwarning(
                "Select Student",
                "Select a student from the table first."
            )
            return

        student_id = self.tree.item(
            selected[0],
            "values"
        )[0]

        data = self.get_form_data()

        if not data["matric_no"] or not data["name"]:
            messagebox.showwarning(
                "Missing Information",
                "Matric number and full name are required."
            )
            return

        try:
            cursor.execute("""
            UPDATE students
            SET matric_no = ?, name = ?, gender = ?,
                department = ?, level = ?, email = ?,
                phone = ?
            WHERE id = ?
            """, (
                data["matric_no"],
                data["name"],
                data["gender"],
                data["department"],
                data["level"],
                data["email"],
                data["phone"],
                student_id
            ))

            conn.commit()

            messagebox.showinfo(
                "Success",
                "Student record updated successfully."
            )

            self.clear_form()
            self.load_students()

        except sqlite3.IntegrityError:
            messagebox.showerror(
                "Error",
                "That matric number belongs to another student."
            )

    # ======================================
    # DELETE STUDENT
    # ======================================

    def delete_student(self):
        selected = self.tree.selection()

        if not selected:
            messagebox.showwarning(
                "Select Student",
                "Select a student from the table first."
            )
            return

        values = self.tree.item(
            selected[0],
            "values"
        )

        student_id = values[0]
        student_name = values[2]

        confirm = messagebox.askyesno(
            "Confirm Deletion",
            f"Delete the record for {student_name}?"
        )

        if confirm:
            cursor.execute("""
            DELETE FROM students WHERE id = ?
            """, (student_id,))

            conn.commit()

            messagebox.showinfo(
                "Success",
                "Student record deleted."
            )

            self.clear_form()
            self.load_students()

    # ======================================
    # CLEAR FORM
    # ======================================

    def clear_form(self):
        for widget in self.fields.values():
            widget.delete(0, tk.END)

        self.search_entry.delete(0, tk.END)

        for item in self.tree.selection():
            self.tree.selection_remove(item)

    # ======================================
    # GENERATE REPORT
    # ======================================

    def generate_report(self):
        cursor.execute("""
        SELECT * FROM students ORDER BY name
        """)

        records = cursor.fetchall()

        if not records:
            messagebox.showinfo(
                "Report",
                "There are no student records to export."
            )
            return

        filename = filedialog.asksaveasfilename(
            defaultextension=".csv",
            filetypes=[("CSV files", "*.csv")],
            initialfile="student_report.csv"
        )

        if not filename:
            return

        try:
            with open(
                filename,
                "w",
                newline="",
                encoding="utf-8-sig"
            ) as file:

                writer = csv.writer(file)

                writer.writerow([
                    "ID",
                    "Matric Number",
                    "Full Name",
                    "Gender",
                    "Department",
                    "Level",
                    "Email",
                    "Phone"
                ])

                writer.writerows(records)

            messagebox.showinfo(
                "Report Generated",
                f"Student report saved successfully.\n\n"
                f"Total students: {len(records)}\n"
                f"Date: {datetime.now():%Y-%m-%d %H:%M}\n"
                f"File: {filename}"
            )

        except OSError as error:
            messagebox.showerror(
                "Export Error",
                f"Could not save the report:\n{error}"
            )

    # ======================================
    # CLEAR WINDOW
    # ======================================

    def clear_window(self):
        for widget in self.root.winfo_children():
            widget.destroy()

# ==========================================
# RUN APPLICATION
# ==========================================

if __name__ == "__main__":
    root = tk.Tk()
    app = StudentManagementSystem(root)

    try:
        root.mainloop()
    finally:
        conn.close()
